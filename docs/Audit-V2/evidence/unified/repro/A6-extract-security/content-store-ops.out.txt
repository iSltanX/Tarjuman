# A6 — عمليات الكاتب من سكربت المحتوى (CL-N1)
STORE_OPS = ["ready","put","delete","orphan","index","settings","prefs","langPrefs","saves","clear"]

## ١. عمليات تُقبل من سكربت المحتوى (العقد يساويه بصفحات الإضافة)
- langPrefs من سكربت المحتوى
  الفعلي:   true
  المتوقع:  "مرفوض (لا يرسله جانب يوتيوب)"
  غير مطابق (يثبت الادعاء)
- saves من سكربت المحتوى
  الفعلي:   true
  المتوقع:  "مرفوض (يكتبه العامل نفسه محليًا)"
  غير مطابق (يثبت الادعاء)
- delete بلا base لفيديو غير المعروض (BBBBBBBBBBB)
  الفعلي:   {"ok":true,"videos":["AAAAAAAAAAA","CCCCCCCCCCC"]}
  المتوقع:  "مرفوض"
  غير مطابق (يثبت الادعاء)
- clear من سكربت المحتوى
  الفعلي:   {"ok":true,"value":{"reinit":"ok","thumbs":null},"keysBefore":15,"keysAfter":["meta"]}
  المتوقع:  "مرفوض"
  غير مطابق (يثبت الادعاء)
  ضابط: مرسِل من موقع آخر يُتجاهل (null): true

## ٢. العمليات التي يرسلها جانب يوتيوب فعلًا (تتبع ديناميكي لأفعال المتحكم وlistIndex)
  العمليات لكل فعل: {"nudgeOffset":["put"],"setDisplayMode":["put"],"updateSettings":["settings"],"updatePrefs":["prefs"],"activateSlot":["put"],"markCue":["put"],"setSlotLang":["put"],"listIndex":["orphan","index"]}
- مجموعة العمليات المرسلة من جانب يوتيوب في هذه التدفقات
  الفعلي:   ["index","orphan","prefs","put","settings"]
  المتوقع:  "ضمن ready وput وorphan وindex وsettings وprefs، بلا clear ولا langPrefs ولا saves ولا delete"
  مطابق
  (ملاحظة: delete لا يصدر من أي فعل في المتحكم؛ كل دوال الكتابة فيه تعيد سجلًا أو undefined لا null.)

## ٣. قائمة السماح المقترحة (محاكاة غلاف حول الخدمة نفسها، دون تعديل المنتج) أمام سياق محتوى مخترق
  clear عبر الغلاف: {"v":1,"ok":false,"code":"TJ_FORBIDDEN_OP"}
- حذف كل فيديو بشرط base بعد قراءة النسخ مباشرة من التخزين (مع قائمة السماح)
  الفعلي:   {"deletes":["AAAAAAAAAAA:true","BBBBBBBBBBB:true","CCCCCCCCCCC:true"],"videosLeft":[]}
  المتوقع:  "قائمة السماح وحدها تمنع ذلك"
  غير مطابق (يثبت الادعاء)
  (والأقصر لسياق مخترق: chrome.storage.local.clear() مباشرة، فلا setAccessLevel في المصدر ومستوى local الافتراضي يتيح سكربتات المحتوى.)

الخلاصة: 5 حالة غير مطابقة للمتوقع (المتوقع: ٤ في القسم ١ تثبت القبول، و١ في القسم ٣ تثبت أن قائمة السماح جزئية؛ والقسم ٢ مطابق).
