## ١. عمليات تُقبل من سكربت المحتوى (العقد يساويه بصفحات الإضافة)
- langPrefs من سكربت المحتوى
  الفعلي:   false
  المتوقع:  "مرفوض (لا يرسله جانب يوتيوب)"
  مطابق
- saves من سكربت المحتوى
  الفعلي:   false
  المتوقع:  "مرفوض (يكتبه العامل نفسه محليًا)"
  مطابق
- delete بلا base لفيديو غير المعروض (BBBBBBBBBBB)
  الفعلي:   {"ok":false,"videos":["AAAAAAAAAAA","BBBBBBBBBBB","CCCCCCCCCCC"]}
  المتوقع:  "مرفوض"
  مطابق
- clear من سكربت المحتوى
  الفعلي:   {"ok":false,"keysBefore":20,"keysAfter":["meta","video:AAAAAAAAAAA","index:AAAAAAAAAAA","bundle:AAAAAAAAAAA","alts:AAAAAAAAAAA","prev:AAAAAAAAAAA","marks:AAAAAAAAAAA","video:BBBBBBBBBBB","index:BBBBBBBBBBB","bundle:BBBBBBBBBBB","alts:BBBBBBBBBBB","prev:BBBBBBBBBBB","marks:BBBBBBBBBBB","video:CCCCCCCCCCC","index:CCCCCCCCCCC","bundle:CCCCCCCCCCC","alts:CCCCCCCCCCC","prev:CCCCCCCCCCC","marks:CCCCCCCCCCC","langPrefs"]}
  المتوقع:  "مرفوض"
  مطابق
  ضابط: مرسِل من موقع آخر يُتجاهل (null): true

## ٢. العمليات التي يرسلها جانب يوتيوب فعلًا (تتبع ديناميكي لأفعال المتحكم وlistIndex)
  العمليات لكل فعل: {"nudgeOffset":["put"],"setDisplayMode":["put"],"updateSettings":["settings"],"updatePrefs":["prefs"],"activateSlot":["put"],"markCue":["put"],"setSlotLang":["put"],"listIndex":["orphan","index"]}
- مجموعة العمليات المرسلة من جانب يوتيوب في هذه التدفقات
  الفعلي:   ["index","orphan","prefs","put","settings"]
  المتوقع:  "ضمن ready وput وorphan وindex وsettings وprefs، بلا clear ولا langPrefs ولا saves ولا delete"
  مطابق
  (ملاحظة: delete لا يصدر من أي فعل في المتحكم؛ كل دوال الكتابة فيه تعيد سجلًا أو undefined لا null.)
