== 6.أ/ب عبر الصفحة: فشل اللغات مرة ثم نجاحها في «أعد المحاولة» ==
التقرير الأول: العنوان وسطر الإعدادات وزر المحاولة
  الفعلي:   {"title":"BK_RESTORE_PARTIAL","settingsLine":null,"retryButton":true}
  المتوقع:  {"title":"BK_RESTORE_PARTIAL","settingsLine":"(سطر يقول إن اللغات وحدها تعذّرت)","retryButton":true}
  مخالف
  نص السطر المعروض: «null»
  القيم المخزّنة فعلًا بعد التقرير الأول (الإعدادات والقاموس)
  الفعلي:   {"offsetStepMs":1000,"terms":3}
  المتوقع:  {"offsetStepMs":1000,"terms":3}
  مطابق
بعد «أعد المحاولة» (نجحت اللغات): العنوان، ومحاولات اللغات، وكتابات الإعدادات، والمخزَّن
  الفعلي:   {"title":"BK_RESTORE_DONE","retryButton":false,"langAttempts":2,"settingsWrites":1,"langStored":{"from":"ja","to":"en"}}
  المتوقع:  {"title":"BK_RESTORE_DONE","retryButton":false,"langAttempts":2,"settingsWrites":1,"langStored":{"from":"ja","to":"en"}}
  مطابق
قسم اللغات في الصفحة نفسها قبل الاستعادة ← بعدها (بلا إعادة تحميل)
  الفعلي:   {"before":{"pair":["تلقائي من مسار الفيديو","العربية"],"remember":true},"after":{"pair":["اليابانية","الإنجليزية"],"remember":false}}
  المتوقع:  {"before":{"pair":["تلقائي من مسار الفيديو","العربية"],"remember":true},"after":{"pair":["اليابانية","الإنجليزية"],"remember":false}}
